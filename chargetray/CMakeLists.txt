cmake_minimum_required(VERSION 3.21)
project(chargetray VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(Qt6 COMPONENTS Test)

add_library(runasync STATIC runasync.cpp)
target_include_directories(runasync PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(runasync PUBLIC Qt6::Core)

add_executable(chargetray chargetray.cpp)
target_link_libraries(chargetray PRIVATE Qt6::Widgets runasync)
target_compile_definitions(chargetray PRIVATE CHARGETRAY_VERSION="${PROJECT_VERSION}")

if(Qt6Test_FOUND)
    enable_testing()
    add_executable(tst_runasync tests/tst_runasync.cpp)
    target_link_libraries(tst_runasync PRIVATE Qt6::Test runasync)
    add_test(NAME runasync COMMAND tst_runasync)
endif()
